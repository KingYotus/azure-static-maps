Click the map to drop a pin
// Initialize when map is ready map.events.add('ready', () => { const dataSource = new atlas.source.DataSource(); map.sources.add(dataSource); // Bubble layer for pins const bubbles = new atlas.layer.BubbleLayer(dataSource, null, { radius: 5, color: '#1a73e8', strokeColor: '#ffffff', strokeWidth: 1 }); map.layers.add(bubbles); // Extract coordinates from click function getLngLatFromClick(e) { if (Array.isArray(e.position) && typeof e.position[0] === 'number' && typeof e.position[1] === 'number' && Math.abs(e.position[0]) <= 180 && Math.abs(e.position[1]) <= 90) { return e.position; } const pixel = e.pixel || e.position; if (Array.isArray(pixel)) return map.pixelsToPositions([pixel])[0]; if (pixel && typeof pixel.x === 'number' && typeof pixel.y === 'number') return map.pixelsToPositions([[pixel.x, pixel.y]])[0]; throw new Error('Unrecognized click event format'); } // Click to add pin map.events.add('click', (e) => { try { const [lng, lat] = getLngLatFromClick(e); dataSource.add(new atlas.data.Point([lng, lat])); updateReadout(lat, lng); } catch (err) { console.error('Click handling failed:', err, e); } }); // Clear pins document.getElementById('btnClear').addEventListener('click', () => { dataSource.clear(); document.getElementById('coordReadout').textContent = 'Pins cleared'; }); // Center to user location document.getElementById('btnMyLocation').addEventListener('click', () => { if (!navigator.geolocation) return alert('Geolocation not supported by this browser.'); navigator.geolocation.getCurrentPosition( pos => { const lat = pos.coords.latitude; const lng = pos.coords.longitude; map.setCamera({ center: [lng, lat], zoom: 14, duration: 500 }); dataSource.add(new atlas.data.Point([lng, lat])); updateReadout(lat, lng); }, err => alert('Unable to get location: ' + err.message), { enableHighAccuracy: true, timeout: 8000, maximumAge: 0 } ); }); // Seed one pin dataSource.add(new atlas.data.Point([-95.9928, 36.1539])); updateReadout(36.1539, -95.9928); }); function updateReadout(lat, lng) { document.getElementById('coordReadout').textContent = `Last pin: lat ${lat.toFixed(5)}, lng ${lng.toFixed(5)}`; console.log(`Pinned at: ${lat}, ${lng}`); }
